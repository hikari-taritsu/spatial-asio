cmake_minimum_required(VERSION 3.24)
project(SpatialASIO VERSION 0.1.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Spatial ASIO requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /permissive- /EHsc /utf-8 /fp:strict)
include_directories(src third_party/asio/common)
set(CORE src/metadata_renderer.cpp src/point_geometry.cpp src/speaker_layout.cpp src/native_spatial.cpp)
set(ASIO src/asio_backend.cpp src/asio_format.cpp)
add_executable(SpatialASIO.Engine src/main.cpp ${CORE} ${ASIO})
target_link_libraries(SpatialASIO.Engine PRIVATE ole32 advapi32 user32 uuid avrt winmm)
include(CTest)
if(BUILD_TESTING)
  add_executable(SpatialASIO.TelemetryTests tests/telemetry_tests.cpp)
  target_link_libraries(SpatialASIO.TelemetryTests PRIVATE ole32)
  add_test(NAME Telemetry COMMAND SpatialASIO.TelemetryTests)
  set_tests_properties(Telemetry PROPERTIES TIMEOUT 5)
  add_executable(SpatialASIO.NativeFixture tests/native_fixture.cpp)
  target_link_libraries(SpatialASIO.NativeFixture PRIVATE ole32 uuid)
  add_executable(SpatialASIO.LayoutTests tests/layout_tests.cpp ${CORE})
  target_link_libraries(SpatialASIO.LayoutTests PRIVATE ole32)
  add_test(NAME Layouts COMMAND SpatialASIO.LayoutTests)
  add_executable(SpatialASIO.Benchmark tests/pipeline_benchmark.cpp ${CORE})
  target_link_libraries(SpatialASIO.Benchmark PRIVATE ole32)
  add_executable(SpatialASIO.CoreTests tests/core_tests.cpp ${CORE})
  target_link_libraries(SpatialASIO.CoreTests PRIVATE ole32)
  add_test(NAME Core COMMAND SpatialASIO.CoreTests)
  add_executable(SpatialASIO.AsioTests tests/asio_runtime_tests.cpp ${ASIO})
  target_compile_definitions(SpatialASIO.AsioTests PRIVATE SPATIAL_AUDIO_ASIO_TESTING)
  target_link_libraries(SpatialASIO.AsioTests PRIVATE ole32 advapi32 user32 uuid)
  add_test(NAME ASIO COMMAND SpatialASIO.AsioTests)
  add_executable(SpatialASIO.OracleCheck tests/oracle_check.cpp ${CORE})
  target_link_libraries(SpatialASIO.OracleCheck PRIVATE ole32)
endif()
